### 2.2 Terminal Risk Calculator

```python
class TerminalRiskCalculator:
    """Burnout prediction from paradox history"""
    
    @staticmethod
    def calculate_terminal_risk(
        paradox_events: List[SoulCradleParadox],
        time_window_days: int = 90,
        viability_threshold: float = 0.3
    ) -> Dict[str, Any]:
        """
        Calculate terminal risk score.
        
        Returns:
            {
                "risk_score": float,
                "risk_level": TerminalRiskLevel,
                "incomplete_resolution_count": int,
                "recommendation": str
            }
        """
        if not paradox_events:
            return {
                "risk_score": 0.0,
                "risk_level": TerminalRiskLevel.LOW,
                "recommendation": "No paradox history. Continue monitoring."
            }
        
        # Filter to time window
        cutoff_date = datetime.now() - timedelta(days=time_window_days)
        recent_paradoxes = [
            p for p in paradox_events 
            if p.timestamp >= cutoff_date
        ]
        
        # Calculate terminal risk
        total_risk = 0.0
        for paradox in recent_paradoxes:
            U = paradox.unresolved_state.unresolved_score
            T = max(paradox.expression_a.tension, paradox.expression_b.tension)
            total_risk += U * T
        
        risk_score = total_risk / len(recent_paradoxes) if recent_paradoxes else 0.0
        
        # Classify risk level
        if risk_score >= 0.7:
            risk_level = TerminalRiskLevel.CRITICAL
        elif risk_score >= 0.4:
            risk_level = TerminalRiskLevel.HIGH
        elif risk_score >= 0.2:
            risk_level = TerminalRiskLevel.MODERATE
        else:
            risk_level = TerminalRiskLevel.LOW
        
        return {
            "risk_score": risk_score,
            "risk_level": risk_level,
            "paradox_count": len(recent_paradoxes),
            "time_window_days": time_window_days,
            "recommendation": _generate_recommendation(risk_level)
        }
```
